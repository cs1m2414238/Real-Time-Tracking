-- DAX Measures Library for Power BI -- 1. Total Fleet Count Total Equipment = COUNTROWS('Fact_Equipment_Inventory') -- 2. Operational Equipment Count Operational Equipment = CALCULATE( COUNTROWS('Fact_Equipment_Inventory'), 'Fact_Equipment_Inventory'[Status] = "Operational" ) -- 3. Readiness Rate % Readiness Rate = DIVIDE( [Operational Equipment], [Total Equipment], 0 ) -- 4. Critical Equipment Flag (<60% Readiness) Critical Equipment = CALCULATE( COUNTROWS('Fact_Equipment_Inventory'), 'Fact_Equipment_Inventory'[Readiness_Pct] < 60 ) -- 5. Total Maintenance Cost Total Maintenance Cost = SUM('Fact_Equipment_Inventory'[Maintenance_Cost_INR]) -- 6. Cost Per Operating Hour Efficiency Cost Per Operating Hour = DIVIDE( [Total Maintenance Cost], SUM('Fact_Equipment_Inventory'[Operating_Hours]), 0 ) -- 7. Upcoming Service in Next 7 Days Upcoming Service 7 Days = CALCULATE( COUNTROWS('Fact_Equipment_Inventory'), 'Fact_Equipment_Inventory'[Next_Service] >= TODAY() && 'Fact_Equipment_Inventory'[Next_Service] <= TODAY() + 7 )
-- 1. Base-Wise Fleet Readiness and Target Variance SELECT b.base_name, b.region, b.target_readiness, COUNT(e.equipment_id) AS total_fleet_size, SUM(CASE WHEN e.status = 'Operational' THEN 1 ELSE 0 END) AS operational_count, ROUND( (SUM(CASE WHEN e.status = 'Operational' THEN 1.0 ELSE 0.0 END) / COUNT(e.equipment_id)) * 100.0, 2 ) AS actual_readiness_pct FROM fact_equipment_inventory e JOIN dim_bases b ON e.base_id = b.base_id GROUP BY b.base_name, b.region, b.target_readiness ORDER BY actual_readiness_pct DESC; -- 2. CTE: High-Risk Problem Asset Identification WITH FleetBenchmarks AS ( SELECT AVG(operating_hours) AS fleet_avg_hours, AVG(maintenance_cost_inr) AS fleet_avg_cost FROM fact_equipment_inventory ) SELECT e.equipment_id, e.base_name, e.equipment_type, e.operating_hours, e.maintenance_cost_inr, e.readiness_pct, e.spare_parts_status FROM fact_equipment_inventory e CROSS JOIN FleetBenchmarks b WHERE e.operating_hours > b.fleet_avg_hours AND e.maintenance_cost_inr > b.fleet_avg_cost AND e.readiness_pct < 65.0 ORDER BY e.readiness_pct ASC; -- 3. Window Function: Priority Ranking within Base SELECT base_name, ROW_NUMBER() OVER (PARTITION BY base_name ORDER BY readiness_pct ASC, downtime_days DESC) AS priority_rank, equipment_id, equipment_type, status, readiness_pct, downtime_days FROM fact_equipment_inventory;
